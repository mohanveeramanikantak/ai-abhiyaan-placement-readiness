# 📘 1. COUNT POSITIVE, NEGATIVE & ZERO

```python
l = [-5, 8, -6, 0, 2, 4, -1]

pos = 0
neg = 0
z = 0

for num in l:
    if num > 0:
        pos += 1
    elif num < 0:
        neg += 1
    else:
        z += 1

print("Positive numbers: ",pos)
print("Negative numbers:" ,neg)
print("Zeroes:" ,z)
